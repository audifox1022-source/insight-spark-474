import { ScatterChart as RechartsScatter, Scatter, XAxis, YAxis, ZAxis, Tooltip, Legend } from 'recharts';

interface ScatterChartProps {
  data: ScatterData[];
  xKey: string;
  yKey: string;
  zKey?: string;
  color?: string;
}

export function ScatterChartComponent({ data, xKey, yKey, zKey, color = '#3b82f6' }: ScatterChartProps) {
  return (
    <RechartsScatter width={400} height={300}>
      <XAxis dataKey={xKey} name="X" />
      <YAxis dataKey={yKey} name="Y" />
      {zKey && <ZAxis dataKey={zKey} range={[40, 400]} name="Z" />}
      <Tooltip cursor={{ strokeDasharray: '3 3' }} />
      <Legend />
      <Scatter name="데이터" data={data} fill={color} />
    </RechartsScatter>
  );
}

### 2.3 실시간 데이터 업데이트

#### 실시간 데이터 소스
```typescript
// src/lib/realtime-data.ts
interface RealtimeDataSource {
  url: string;
  interval?: number;
  transform?: (data: any) => any;
}

export function useRealtimeData(source: RealtimeDataSource) {
  const [data, setData] = useState<any>(null);
  const [status, setStatus] = useState<'connecting' | 'connected' | 'error'>('connecting');
  
  useEffect(() => {
    const ws = new WebSocket(source.url);
    
    ws.onopen = () => setStatus('connected');
    ws.onmessage = (event) => {
      const transformed = source.transform 
        ? source.transform(JSON.parse(event.data))
        : JSON.parse(event.data);
      setData(transformed);
    };
    ws.onerror = () => setStatus('error');
    ws.onclose = () => setStatus('connecting');
    
    return () => ws.close();
  }, [source.url]);
  
  return { data, status };
}
```

### 2.5 데이터 시각화 자동 추천

#### AI 기반 차트 추천
```typescript
// src/lib/chart-recommendation.ts
interface DataAnalysis {
  columns: { name: string; type: 'numeric' | 'categorical' | 'temporal' }[];
  rowCount: number;
  valueRange?: { min: number; max: number };
}

export function recommendChartType(data: DataAnalysis): ChartType[] {
  const recommendations: ChartType[] = [];
  
  // 수치형 데이터가 2개 이상
  const numericCols = data.columns.filter(c => c.type === 'numeric');
  if (numericCols.length >= 2) {
    recommendations.push('scatter');
    recommendations.push('bubble');
  }
  
  // 시계열 데이터
  const temporalCols = data.columns.filter(c => c.type === 'temporal');
  if (temporalCols.length > 0) {
    recommendations.push('line');
    recommendations.push('area');
  }
  
  // 범주형 데이터
  const categoricalCols = data.columns.filter(c => c.type === 'categorical');
  if (categoricalCols.length > 0 && numericCols.length > 0) {
    recommendations.push('bar');
    recommendations.push('pie');
  }
  
  // 기본 추천
  if (recommendations.length === 0) {
    recommendations.push('bar');
    recommendations.push('line');
  }
  
  return recommendations;
}
```
