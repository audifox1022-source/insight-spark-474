#### AI 제안 구현
```typescript
// src/lib/ai-suggestions.ts
interface SuggestionContext {
  currentSlide: Slide;
  previousSlide?: Slide;
  nextSlide?: Slide;
  presentationTheme: string;
  audienceFocus: string;
}

interface ContentSuggestion {
  type: 'heading' | 'description' | 'layout' | 'data' | 'citation';
  confidence: number;
  content: string;
  reason: string;
}

export async function generateContentSuggestions(
  context: SuggestionContext
): Promise<ContentSuggestion[]> {
  const prompt = buildSuggestionPrompt(context);
  const response = await callGeminiAPI(SUGGESTION_SYSTEM_PROMPT, prompt);
  return parseSuggestions(response);
}

function buildSuggestionPrompt(context: SuggestionContext): string {
  return `
현재 슬라이드: ${JSON.stringify(context.currentSlide)}
이전 슬라이드: ${context.previousSlide ? JSON.stringify(context.previousSlide) : '없음'}
다음 슬라이드: ${context.nextSlide ? JSON.stringify(context.nextSlide) : '없음'}
발표 테마: ${context.presentationTheme}
청중: ${context.audienceFocus}

이 슬라이드를 개선할 수 있는 구체적인 제안을 제공해주세요.
`;
}
```

#### 제안 UI 컴포넌트
```typescript
// src/components/AISuggestionPanel.tsx
interface AISuggestionPanelProps {
  suggestions: ContentSuggestion[];
  onApply: (suggestion: ContentSuggestion) => void;
  onDismiss: (index: number) => void;
}

export function AISuggestionPanel({ 
  suggestions, onApply, onDismiss 
}: AISuggestionPanelProps) {
  return (
    <div className="space-y-2">
      {suggestions.map((suggestion, index) => (
        <SuggestionCard
          key={index}
          suggestion={suggestion}
          onApply={() => onApply(suggestion)}
          onDismiss={() => onDismiss(index)}
        />
      ))}
    </div>
  );
}

function SuggestionCard({ 
  suggestion, onApply, onDismiss 
}: {
  suggestion: ContentSuggestion;
  onApply: () => void;
  onDismiss: () => void;
}) {
  return (
    <div className="p-3 rounded-lg border bg-card">
      <div className="flex items-center justify-between mb-2">
        <Badge variant={suggestion.type === 'heading' ? 'default' : 'secondary'}>
          {suggestion.type}
        </Badge>
        <span className="text-xs text-muted-foreground">
          {Math.round(suggestion.confidence * 100)}% 확신
        </span>
      </div>
      <p className="text-sm mb-2">{suggestion.content}</p>
      <p className="text-xs text-muted-foreground mb-3">{suggestion.reason}</p>
      <div className="flex gap-2">
        <Button size="sm" onClick={onApply}>적용</Button>
        <Button size="sm" variant="ghost" onClick={onDismiss}>무시</Button>
      </div>
    </div>
  );
}
```

```typescript
// src/lib/smart-editing.ts
interface EditingSuggestion {
  type: 'typo' | 'grammar' | 'style' | 'consistency';
  original: string;
  corrected: string;
  explanation: string;
  confidence: number;
}

export async function checkSmartEditing(
  content: string,
  context: {
    brandGuide?: string;
    terminology?: Record<string, string>;
    tone?: string;
  }
): Promise<EditingSuggestion[]> {
  const prompt = buildEditingCheckPrompt(content, context);
  const response = await callGeminiAPI(EDITING_CHECK_SYSTEM_PROMPT, prompt);
  return parseEditingSuggestions(response);
}

export function applyEditingSuggestion(
  content: string,
  suggestion: EditingSuggestion
): string {
  return content.replace(suggestion.original, suggestion.corrected);
}
```

```typescript
// src/components/QualityMetrics.tsx
interface QualityMetricsProps {
  slide: Slide;
  metrics: {
    readability: number;
    professionalism: number;
    visualization: number;
    consistency: number;
  };
}

export function QualityMetrics({ slide, metrics }: QualityMetricsProps) {
  return (
    <div className="flex gap-4 p-3 rounded-lg bg-muted/50">
      <MetricBadge 
        label="가독성" 
        value={metrics.readability} 
        icon={<BookOpen className="w-4 h-4" />} 
      />
      <MetricBadge 
        label="전문성" 
        value={metrics.professionalism} 
        icon={<Award className="w-4 h-4" />} 
      />
      <MetricBadge 
        label="시각화" 
        value={metrics.visualization} 
        icon={<BarChart3 className="w-4 h-4" />} 
      />
      <MetricBadge 
        label="일관성" 
        value={metrics.consistency} 
        icon={<CheckCircle2 className="w-4 h-4" />} 
      />
    </div>
  );
}

function MetricBadge({ 
  label, value, icon 
}: {
  label: string;
  value: number;
  icon: React.ReactNode;
}) {
  const getColor = (v: number) => {
    if (v >= 80) return 'text-emerald-500';
    if (v >= 60) return 'text-amber-500';
    return 'text-red-500';
  };
  
  return (
    <div className="flex items-center gap-2">
      <span className={getColor(value)}>{icon}</span>
      <div className="text-xs">
        <span className="text-muted-foreground">{label}</span>
        <span className={`ml-1 font-bold ${getColor(value)}`}>{value}</span>
      </div>
    </div>
  );
}
```
