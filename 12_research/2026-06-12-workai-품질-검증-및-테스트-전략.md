```typescript
// src/services/ai/__tests__/geminiService.test.ts
import { describe, it, expect, vi } from 'vitest';
import { extractJson, aiService } from '../geminiService';

describe('extractJson', () => {
  it('should parse valid JSON', () => {
    const input = '{"slides": [{"title": "Test"}]}';
    const result = extractJson(input);
    expect(result).toEqual([{"title": "Test"}]);
  });

  it('should handle markdown code blocks', () => {
    const input = '```json\n{"slides": []}\n```';
    const result = extractJson(input);
    expect(result).toEqual([]);
  });

  it('should repair trailing commas', () => {
    const input = '{"slides": [{"title": "Test",},]}';
    const result = extractJson(input);
    expect(result).toBeDefined();
  });

  it('should return null for invalid JSON', () => {
    const input = 'not json';
    const result = extractJson(input);
    expect(result).toBeNull();
  });
});

describe('aiService', () => {
  it('should generate presentation', async () => {
    const mockResponse = { slides: [] };
    vi.mock('../api-client', () => ({
      callGeminiAPI: vi.fn().mockResolvedValue(JSON.stringify(mockResponse))
    }));
    
    const result = await aiService.generatePresentation({
      fileData: 'test',
      template: 'auto',
      settings: { slideCount: 10 }
    });
    
    expect(result).toBeDefined();
  });
});
```

#### 라이브러리 함수 테스트
```typescript
// src/lib/__tests__/deck-quality-audit.test.ts
import { describe, it, expect } from 'vitest';
import { auditPresentationQuality } from '../deck-quality-audit';

describe('auditPresentationQuality', () => {
  it('should return D grade for empty presentation', () => {
    const result = auditPresentationQuality(null);
    expect(result.score).toBe('D');
    expect(result.scoreValue).toBe(0);
  });

  it('should detect missing cover slide', () => {
    const presentation = {
      slides: [
        { layout: 'default', content: [] }
      ]
    };
    const result = auditPresentationQuality(presentation);
    expect(result.improvements).toContainEqual(
      expect.objectContaining({ category: 'Layout' })
    );
  });

  it('should give high score for well-structured presentation', () => {
    const presentation = {
      slides: [
        { layout: 'cover', title: 'Title', content: [] },
        { layout: 'chart', title: 'Data', content: [{ heading: 'H', description: 'D' }] },
        { layout: 'timeline', title: 'Plan', content: [{ heading: 'H', description: 'D' }] },
        { layout: 'summary', title: 'Conclusion', content: [{ heading: 'H', description: 'D' }] }
      ]
    };
    const result = auditPresentationQuality(presentation);
    expect(result.scoreValue).toBeGreaterThanOrEqual(80);
  });
});
```

```typescript
// src/components/__tests__/PresentationSetupForm.test.tsx
import { describe, it, expect, vi } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import { PresentationSetupForm } from '../PresentationSetupForm';

describe('PresentationSetupForm', () => {
  const defaultProps = {
    info: { title: 'Test', objective: 'Test', audience: 'Test' },
    onChange: vi.fn(),
    settings: { difficulty: 'medium', slideCount: 10 },
    onSettingsChange: vi.fn(),
    onGenerate: vi.fn(),
    onBack: vi.fn(),
    isGenerating: false,
  };

  it('should render form fields', () => {
    render(<PresentationSetupForm {...defaultProps} />);
    expect(screen.getByLabelText('발표 제목')).toBeInTheDocument();
    expect(screen.getByLabelText('목표/결정 요청')).toBeInTheDocument();
  });

  it('should call onGenerate when button clicked', () => {
    render(<PresentationSetupForm {...defaultProps} />);
    fireEvent.click(screen.getByText('이 설정으로 발표자료 생성하기'));
    expect(defaultProps.onGenerate).toHaveBeenCalled();
  });

  it('should disable button when generating', () => {
    render(<PresentationSetupForm {...defaultProps} isGenerating={true} />);
    expect(screen.getByText(/AI가 발표자료를 설계하고 있습니다/)).toBeDisabled();
  });
});
```

### 2.4 통합 테스트

#### 생성 플로우 통합 테스트
```typescript
// src/__tests__/generation-flow.test.tsx
import { describe, it, expect, vi } from 'vitest';
import { render, screen, waitFor } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import App from '../App';

describe('Presentation Generation Flow', () => {
  it('should complete full generation flow', async () => {
    const user = userEvent.setup();
    render(<App />);
    
    // 1. 파일 업로드 또는 주제 입력
    const textarea = screen.getByPlaceholderText(/발표 주제/);
    await user.type(textarea, 'AI 트렌드 보고서');
    
    // 2. 설정 단계로 이동
    await user.click(screen.getByText('설정 단계로 이동'));
    
    // 3. 설정 완료 후 생성
    await user.click(screen.getByText('이 설정으로 발표자료 생성하기'));
    
    // 4. 생성 완료 대기
    await waitFor(() => {
      expect(screen.getByText('발표자료 생성 완료')).toBeInTheDocument();
    });
  });
});
```
